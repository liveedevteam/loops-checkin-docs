// createBooking.js — atomic seat hold (no race condition possible)
const round = await Round.findOneAndUpdate(
  {
    _id: roundID,
    // Atomic guard: only update if seats are still available
    $expr: {
      $lte: [
        { $add: ['$seats', '$tempSeats', requestedSeats] },
        '$maxSeats'
      ]
    }
  },
  { $inc: { tempSeats: requestedSeats }, updateDate: dayjs.tz().format() },
  { new: true, session }
).exec()

if (!round) {
